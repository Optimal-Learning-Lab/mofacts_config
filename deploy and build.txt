if (!(Test-Path "$SamlDir\memphis-saml-public-cert.pem")) { throw "Missing Memphis SAML public cert" }
if (!(Test-Path "$SamlDir\memphis-saml-private-key.pem")) { throw "Missing Memphis SAML private key" }

$RequiredStagingEnv = @(
  "METEOR_SETTINGS_HOST_PATH",
  "MOFACTS_ENV_FILE_HOST_PATH",
  "ROOT_URL",
  "MONGO_URL",
  "EXPECTED_MONGO_DB_NAME",
  "MONGO_INITDB_ROOT_USERNAME",
  "MONGO_INITDB_ROOT_PASSWORD",
  "MOFACTS_MONGO_APP_USERNAME",
  "MOFACTS_MONGO_APP_PASSWORD",
  "MOFACTS_MONGO_REPLICA_SET_NAME",
  "MOFACTS_MONGO_REPLICA_SET_MEMBER",
  "MONGO_REPLICA_SET_KEYFILE_HOST_PATH",
  "MOFACTS_REDIS_PASSWORD"
  "MOFACTS_SECURITY_AUDIT_INGEST_SECRET"
)
$StagingEnvKeys = Select-String -Path $StageEnvPath -Pattern "^[A-Za-z_][A-Za-z0-9_]*=" | ForEach-Object { ($_.Line -split "=", 2)[0] }
$MissingStagingEnv = $RequiredStagingEnv | Where-Object { $StagingEnvKeys -notcontains $_ }
if ($MissingStagingEnv) { throw "Missing staging env keys: $($MissingStagingEnv -join ', ')" }

$StagingAuditSecret = ($StagingAuditSecretLine -split '=', 2)[1]
if ([string]::IsNullOrWhiteSpace($StagingAuditSecret) -or $StagingAuditSecret.Length -lt 32) { throw "Staging security audit ingestion secret must contain at least 32 characters" }

# Validate the exact staging definition, then build and push that app image.
docker compose --env-file $StageEnvPath -f .\docker-compose-staging.yml config --quiet
docker compose --env-file $StageEnvPath -f .\docker-compose-staging.yml build --no-cache mofacts
docker compose --env-file $StageEnvPath -f .\docker-compose-staging.yml push mofacts

# Create the runtime filesystem required by the compose mounts.
ssh -i $StageKey $StageHost "command -v docker >/dev/null && docker compose version >/dev/null && sudo mkdir -p /var/www/mofacts/mongo-init /var/www/mofacts/mongodb /etc/mofacts /backups/mofacts /mofactsAssets_override /dynamic-assets && sudo chown -R ubuntu:ubuntu /var/www/mofacts /backups/mofacts"

scp -i $StageKey .\docker-compose-staging.yml "${StageHost}:/var/www/mofacts/docker-compose.yaml"
scp -i $StageKey $StageEnvPath "${StageHost}:/var/www/mofacts/.env"

# Preflight and start the app stack.
ssh -i $StageKey $StageHost "test -f /var/www/mofacts/docker-compose.yaml && test -f /var/www/mofacts/.env && test -f /var/www/mofacts/settingsstaging.json && test -f /var/www/mofacts/mongo-init/01-create-mofacts-app-user.js && test -f /var/www/mofacts/mongodb/entrypoint.sh && sudo test -s /etc/mofacts/mongodb-keyfile && test -f /mofactsAssets_override/memphis-saml-public-cert.pem && test -f /mofactsAssets_override/memphis-saml-private-key.pem && test -d /dynamic-assets && cd /var/www/mofacts && sudo docker compose --env-file .env -f docker-compose.yaml config --quiet"

ssh -i $StageKey $StageHost "cd /var/www/mofacts-mcp-sidecar && sudo docker compose -p mofacts-mcp-staging -f docker-compose.yml -f docker-compose.staging-server.yml up -d --build"

$ProdImageRepository = "ppavlikmemphis/mofacts-mini"

cd $DeployDir

if (!(Test-Path $ProdKey)) { throw "Missing production SSH key: $ProdKey" }
if (!(Test-Path $ProdReplicaKey)) { throw "Missing production MongoDB replica-set keyfile: $ProdReplicaKey" }
if (!(Test-Path .\docker-compose.yml)) { throw "Missing production compose file" }
if (!(Test-Path $ProdChangeStreamsComposePath)) { throw "Missing production Change Streams compose overlay: $ProdChangeStreamsComposePath" }
if (!(Test-Path $ProdEnvPath)) { throw "Missing production env file: $ProdEnvPath" }
if (!(Test-Path $ProdSettingsPath)) { throw "Missing production settings file: $ProdSettingsPath" }
if (!(Test-Path $ProdApacheTlsSitePath)) { throw "Missing production Apache TLS site: $ProdApacheTlsSitePath" }
if (!(Test-Path .\mongo-init\01-create-mofacts-app-user.js)) { throw "Missing Mongo app-user init script" }
if (!(Test-Path "$SamlDir\memphis-saml-public-cert.pem")) { throw "Missing Memphis SAML public cert" }
if (!(Test-Path "$SamlDir\memphis-saml-private-key.pem")) { throw "Missing Memphis SAML private key" }

$RequiredProductionEnv = @(
  "METEOR_SETTINGS_HOST_PATH",
  "MOFACTS_ENV_FILE_HOST_PATH",
  "ROOT_URL",
  "MONGO_URL",
  "EXPECTED_MONGO_DB_NAME",
  "METEOR_REACTIVITY_ORDER",
  "DDP_TRANSPORT",
  "MONGO_INITDB_ROOT_USERNAME",
  "MONGO_INITDB_ROOT_PASSWORD",
  "MOFACTS_MONGO_APP_USERNAME",
  "MOFACTS_MONGO_APP_PASSWORD",
  "MOFACTS_MONGO_REPLICA_SET_NAME",
  "MOFACTS_MONGO_REPLICA_SET_MEMBER",
  "MONGO_REPLICA_SET_KEYFILE_HOST_PATH",
  "MOFACTS_BACKUP_HOST_PATH",
  "MOFACTS_REDIS_PASSWORD",
  "MOFACTS_SECURITY_AUDIT_INGEST_SECRET",
  "DOCKER_REGISTRY",
  "IMAGE_NAME",
  "IMAGE_TAG"
)
$ProductionEnvKeys = Select-String -Path $ProdEnvPath -Pattern "^[A-Za-z_][A-Za-z0-9_]*=" | ForEach-Object { ($_.Line -split "=", 2)[0] }
$MissingProductionEnv = $RequiredProductionEnv | Where-Object { $ProductionEnvKeys -notcontains $_ }
if ($MissingProductionEnv) { throw "Missing production env keys: $($MissingProductionEnv -join ', ')" }
$ProductionEnvValues = @{}
Get-Content $ProdEnvPath | Where-Object { $_ -match "^[A-Za-z_][A-Za-z0-9_]*=" } | ForEach-Object {
  $Name, $Value = $_ -split "=", 2
  $ProductionEnvValues[$Name] = $Value
}
if ("$($ProductionEnvValues['DOCKER_REGISTRY'])/$($ProductionEnvValues['IMAGE_NAME'])" -ne $ProdImageRepository) { throw "Production image repository does not match the recorded release" }
if ($ProductionEnvValues['METEOR_REACTIVITY_ORDER'] -ne 'changeStreams') { throw "Production reactivity order must be changeStreams" }
if ([string]::IsNullOrWhiteSpace($ProductionEnvValues['MOFACTS_SECURITY_AUDIT_INGEST_SECRET']) -or $ProductionEnvValues['MOFACTS_SECURITY_AUDIT_INGEST_SECRET'].Length -lt 32) { throw "Production security audit ingestion secret must contain at least 32 characters" }
$ProdApacheTlsSiteText = Get-Content -LiteralPath $ProdApacheTlsSitePath -Raw
$ProdCspHeaderMatches = [regex]::Matches($ProdApacheTlsSiteText, '(?m)^[ \t]*Header always set Content-Security-Policy "([^"]+)"[ \t]*$')
if ($ProdCspHeaderMatches.Count -ne 1) { throw "Production Apache TLS site must define exactly one enforced CSP header" }
$ExpectedProductionCsp = $ProdCspHeaderMatches[0].Groups[1].Value
# The app's MongoDB driver validates the opaque URI, selected database,
# authentication, and topology during startup. Do not parse or print the URI.
# Before this procedure, set MONGO_REPLICA_SET_KEYFILE_HOST_PATH to the remote
# installed path /etc/mofacts/mongodb-keyfile, choose the immutable replica-set
# name/member, and update MONGO_URL to the same set identity and authSource.

# Validate the exact production definition, then build and push that app image.
docker compose --env-file $ProdEnvPath -f .\docker-compose.yml -f $ProdChangeStreamsComposePath config --quiet
docker compose --env-file $ProdEnvPath -f .\docker-compose.yml build --no-cache mofacts
docker compose --env-file $ProdEnvPath -f .\docker-compose.yml push mofacts
$PublishedImageManifest = docker buildx imagetools inspect "$($ProdImageRepository):$ProdImageTag" --format '{{json .Manifest}}' | ConvertFrom-Json
$ProdImageDigest = $PublishedImageManifest.digest
if ([string]::IsNullOrWhiteSpace($ProdImageDigest) -or $ProdImageDigest -notmatch '^sha256:[0-9a-f]{64}$') { throw "Registry did not return an immutable production image digest" }

# Only after the push succeeds, keep the authoritative production environment
# pointed at the published image. This changes IMAGE_TAG and preserves all
# secret values in the Desktop file.
$ProdEnvText = Get-Content -LiteralPath $ProdEnvPath -Raw
if ($ProdEnvText -notmatch '(?m)^IMAGE_TAG=') { throw "Production env file has no IMAGE_TAG entry" }
$ProdEnvText = [regex]::Replace($ProdEnvText, '(?m)^IMAGE_TAG=.*$', "IMAGE_TAG=$ProdImageTag")
Set-Content -LiteralPath $ProdEnvPath -Value $ProdEnvText -NoNewline
Write-Host "Published release commit: $ProdReleaseCommit"
Write-Host "Included uncommitted local changes: $ProdWorkingTreeDirty"
Write-Host "Published image: $($ProdImageRepository):$ProdImageTag"
Write-Host "Published digest: $ProdImageDigest"

# Create the runtime filesystem required by the compose mounts.
ssh -i $ProdKey $ProdHost "command -v docker >/dev/null && docker compose version >/dev/null && sudo mkdir -p /var/www/mofacts/mongo-init /var/www/mofacts/mongodb /etc/mofacts /backups/mofacts /mofactsAssets_override /dynamic-assets && sudo chown -R ubuntu:ubuntu /var/www/mofacts /backups/mofacts"

# Copy app runtime inputs.
scp -i $ProdKey .\docker-compose.yml "${ProdHost}:/var/www/mofacts/docker-compose.yaml"
scp -i $ProdKey $ProdChangeStreamsComposePath "${ProdHost}:/var/www/mofacts/docker-compose.change-streams.yaml"
scp -i $ProdKey $ProdEnvPath "${ProdHost}:/var/www/mofacts/.env"
scp -i $ProdKey $ProdSettingsPath "${ProdHost}:/var/www/mofacts/settings.json"
scp -i $ProdKey $ProdApacheTlsSitePath "${ProdHost}:/tmp/000-default-le-ssl.conf"
scp -i $ProdKey .\mongo-init\01-create-mofacts-app-user.js "${ProdHost}:/var/www/mofacts/mongo-init/01-create-mofacts-app-user.js"
scp -i $ProdKey .\mongodb\* "${ProdHost}:/var/www/mofacts/mongodb/"
scp -i $ProdKey .\backup-self-hosted.sh "${ProdHost}:/var/www/mofacts/backup-self-hosted.sh"
scp -i $ProdKey .\restore-self-hosted.sh "${ProdHost}:/var/www/mofacts/restore-self-hosted.sh"
scp -i $ProdKey $ProdReplicaKey "${ProdHost}:/tmp/production-mongodb-keyfile"
ssh -i $ProdKey $ProdHost "sudo install -o root -g root -m 600 /tmp/production-mongodb-keyfile /etc/mofacts/mongodb-keyfile && rm -f /tmp/production-mongodb-keyfile && chmod 600 /var/www/mofacts/settings.json && chmod 755 /var/www/mofacts/mongodb/*.sh /var/www/mofacts/backup-self-hosted.sh /var/www/mofacts/restore-self-hosted.sh"

# Install Memphis SAML cert/key.
scp -i $ProdKey "$SamlDir\memphis-saml-public-cert.pem" "${ProdHost}:/tmp/memphis-saml-public-cert.pem"
scp -i $ProdKey "$SamlDir\memphis-saml-private-key.pem" "${ProdHost}:/tmp/memphis-saml-private-key.pem"
ssh -i $ProdKey $ProdHost "sudo mv /tmp/memphis-saml-public-cert.pem /mofactsAssets_override/memphis-saml-public-cert.pem && sudo mv /tmp/memphis-saml-private-key.pem /mofactsAssets_override/memphis-saml-private-key.pem && sudo chown root:root /mofactsAssets_override/memphis-saml-*.pem && sudo chmod 644 /mofactsAssets_override/memphis-saml-public-cert.pem && sudo chmod 600 /mofactsAssets_override/memphis-saml-private-key.pem"

# Preflight the complete production configuration without starting or replacing services.
ssh -i $ProdKey $ProdHost "test -f /var/www/mofacts/docker-compose.yaml && test -f /var/www/mofacts/docker-compose.change-streams.yaml && test -f /var/www/mofacts/.env && test -f /var/www/mofacts/settings.json && test -f /tmp/000-default-le-ssl.conf && test -f /var/www/mofacts/mongo-init/01-create-mofacts-app-user.js && test -f /var/www/mofacts/mongodb/entrypoint.sh && test -f /var/www/mofacts/mongodb/init-replica-set.sh && sudo test -s /etc/mofacts/mongodb-keyfile && test -x /var/www/mofacts/backup-self-hosted.sh && test -x /var/www/mofacts/restore-self-hosted.sh && test -f /mofactsAssets_override/memphis-saml-public-cert.pem && test -f /mofactsAssets_override/memphis-saml-private-key.pem && test -d /dynamic-assets && cd /var/www/mofacts && sudo docker compose --env-file .env -f docker-compose.yaml -f docker-compose.change-streams.yaml config --quiet"

# Deploy this source-built image with the existing production replica set and
# the production Change Streams overlay. This is the ordinary app deployment;
# it does not repeat the completed database-topology conversion.
ssh -i $ProdKey $ProdHost "cd /var/www/mofacts && sudo docker compose --env-file .env -f docker-compose.yaml -f docker-compose.change-streams.yaml pull mofacts && sudo docker compose --env-file .env -f docker-compose.yaml -f docker-compose.change-streams.yaml up -d --no-build --wait mofacts && sudo docker compose --env-file .env -f docker-compose.yaml -f docker-compose.change-streams.yaml ps"
$RunningProductionImage = (ssh -i $ProdKey $ProdHost "sudo docker inspect mofacts --format '{{.Config.Image}}'").Trim()

$RunningProductionRevision = $RunningProductionLabels['org.opencontainers.image.revision']
if ($RunningProductionRevision -ne $ProdReleaseCommit) { throw "Production image source-revision label does not match the deployed release" }

# Install the tracked production TLS virtual host. Preserve the active file and
# restore it automatically if Apache validation or reload fails.

$ProductionHeadResponse = Invoke-WebRequest -Uri "https://mofacts.optimallearning.org/admin/security-audits" -Method Head
$LiveProductionCsp = [string]$ProductionHeadResponse.Headers['Content-Security-Policy']
if ($LiveProductionCsp -ne $ExpectedProductionCsp) { throw "Production CSP header does not match the tracked Apache TLS site" }

# Automatically record the successfully deployed immutable release beside the
# authoritative Desktop inputs. Keep verification and recording in one
# statement so a failed deployment cannot be recorded by continuing below an
# earlier exception.

  [string]::IsNullOrWhiteSpace($ProdImageDigest) -or
  $ProdImageDigest -notmatch '^sha256:[0-9a-f]{64}$' -or
  $RunningProductionImage -ne "$($ProdImageRepository):$ProdImageTag" -or
  $RunningProductionRevision -ne $ProdReleaseCommit -or
  $LiveProductionCsp -ne $ExpectedProductionCsp
) {
  throw "Production deployment is not verified; release record was not written"
} else {

    deployedAtUtc = [DateTime]::UtcNow.ToString('o')
    baseCommit = $ProdReleaseCommit
    includedUncommittedLocalChanges = $ProdWorkingTreeDirty
    image = "$($ProdImageRepository):$ProdImageTag"
    tag = $ProdImageTag
    digest = $ProdImageDigest
  } | ConvertTo-Json | Set-Content -LiteralPath $ProdReleaseRecordPath
