# Paste these commands in PowerShell.

# Lines inside ssh "..." run on Linux. Keep those as plain commands.

# Run app Docker commands from:
#   C:\dev\MoFaCTS\mofacts\.deploy
#
# Run MCP sidecar local Docker commands from:
#   C:\Users\ppavl\OneDrive\Active projects\mofacts_config\mofacts-mcp-sidecar

# App files on staging and remote/production live under:
#   /var/www/mofacts

# MCP sidecar files on staging and remote/production live under:
#   /var/www/mofacts-mcp-sidecar
#
# Important:
# - The app can be deployed by itself.
# - MCP sidecar deploy requires its compose files, services/playwright-mcp Dockerfile,
#   and services/mongo-mcp source files to be copied first.
# - Do not run the MCP remote docker compose command until docker-compose.yml,
#   docker-compose.remote-server.yml, services/playwright-mcp, and services/mongo-mcp exist on the remote server.
# - Do not recursively scp the whole services/mongo-mcp folder. It may contain
#   node_modules, which transfers thousands of unnecessary files. Copy only the
#   Docker build inputs listed below.
# - Staging currently uses .env.production in the commands below because that is how this
#   sheet was previously written. If staging should use .env.staging, switch both the
#   build/push env file and the scp source for staging.


# ---------------------------------------------------------------------------
# Local app rebuild with current code
# ---------------------------------------------------------------------------

cd C:\dev\MoFaCTS\mofacts\.deploy

docker compose --env-file .env.local -f docker-compose.yml -f docker-compose.local.yml build --no-cache mofacts
docker compose --env-file .env.local -f docker-compose.yml -f docker-compose.local.yml up -d mofacts


# ---------------------------------------------------------------------------
# Local MCP sidecar rebuild
# ---------------------------------------------------------------------------

cd "C:\Users\ppavl\OneDrive\Active projects\mofacts_config\mofacts-mcp-sidecar"

docker compose -f docker-compose.yml -f docker-compose.local-server.yml down --remove-orphans
docker compose -f docker-compose.yml -f docker-compose.local-server.yml up --build

# Performance notes:
# - Meteor MergeBox work can dominate CPU when many live subscriptions watch data
#   that changes frequently.
# - Trial payload JSON parsing/stringifying blocks the Node event loop while it runs.
# - JSON-based deep cloning, such as JSON.parse(JSON.stringify(...)), is expensive
#   in hot paths.
# - Mongo writes trigger Meteor observer/oplog work, so frequent trial writes can
#   cascade into subscription invalidation and update checks.
