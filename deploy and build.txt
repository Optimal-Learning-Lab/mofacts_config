MoFaCTS deploy/build cheat sheet

# Paste these commands in PowerShell.
# PowerShell comments start with #, so comment lines are safe to paste.
# Lines inside ssh "..." run on Linux. Keep those as plain commands.
#
# This is the private operator cheat sheet for the current hosted MoFaCTS
# deployment. Public/open-core deploy docs live in:
#   C:\dev\MoFaCTS\docs\deployment
#
# Architecture reminders:
# - App deploy and MCP sidecar deploy are separate tasks.
# - H5P content is uploaded as content packages. The H5P tester .h5p files in
#   this repo bundle their required H5P libraries; do not seed H5P libraries
#   during app deploy.
# - The remote /h5p-content and /h5p-libraries directories are runtime storage
#   mounts used by the app after packages are uploaded.


# ---------------------------------------------------------------------------
# Local app loops
# ---------------------------------------------------------------------------

# Fast local UI/app loop at http://localhost:3200.
cd C:\dev\MoFaCTS\deploy
.\hotfix-dev.ps1 start
.\hotfix-dev.ps1 status

# Production-shaped local app-code verification without pushing an image.
cd C:\dev\MoFaCTS\deploy
.\hotfix-local.ps1

# Full local Docker image check at http://localhost:3100.
# Use only when you deliberately want to verify the Docker image path.
cd C:\dev\MoFaCTS\deploy
docker compose --env-file .env.local -f docker-compose.yml -f docker-compose.local.yml build --no-cache mofacts
docker compose --env-file .env.local -f docker-compose.yml -f docker-compose.local.yml up -d mofacts


# ---------------------------------------------------------------------------
# Local MCP sidecar
# ---------------------------------------------------------------------------

# MCP sidecar pointed at the normal hotfix-dev app on http://localhost:3200.
cd C:\dev\MoFaCTS\mofacts-mcp-sidecar
docker compose -f docker-compose.yml -f docker-compose.hotfix-dev.yml down --remove-orphans
docker compose -f docker-compose.yml -f docker-compose.hotfix-dev.yml up --build

# MCP sidecar pointed at the Dockerized local app on http://localhost:3100.
cd C:\dev\MoFaCTS\mofacts-mcp-sidecar
docker compose -f docker-compose.yml -f docker-compose.local-server.yml down --remove-orphans
docker compose -f docker-compose.yml -f docker-compose.local-server.yml up --build


# ---------------------------------------------------------------------------
# Staging app deploy
# ---------------------------------------------------------------------------

cd C:\dev\MoFaCTS\deploy

$StageKey = "C:\Users\ppavl\OneDrive\Desktop\mykey.pem"
$StageHost = "ubuntu@44.253.109.187"
$SamlDir = "C:\Users\ppavl\OneDrive\Active projects\mofacts-saml"

if (!(Test-Path .\docker-compose-staging.yml)) { throw "Missing staging compose file" }
if (!(Test-Path .\.env.staging)) { throw "Missing staging env file" }
if (!(Test-Path .\settingsstaging.json)) { throw "Missing staging settings file" }
if (!(Test-Path "$SamlDir\memphis-saml-public-cert.pem")) { throw "Missing Memphis SAML public cert" }
if (!(Test-Path "$SamlDir\memphis-saml-private-key.pem")) { throw "Missing Memphis SAML private key" }

# Build and push the staging app image.
docker compose --env-file .\.env.staging build --no-cache mofacts
docker compose --env-file .\.env.staging push mofacts

# Create the runtime filesystem required by the compose mounts.
ssh -i $StageKey $StageHost "command -v docker >/dev/null && docker compose version >/dev/null && sudo mkdir -p /var/www/mofacts /mofactsAssets_override /dynamic-assets /h5p-content /h5p-libraries && sudo chown -R ubuntu:ubuntu /var/www/mofacts"

# Copy app runtime inputs.
scp -i $StageKey .\docker-compose-staging.yml "${StageHost}:/var/www/mofacts/docker-compose.yaml"
scp -i $StageKey .\.env.staging "${StageHost}:/var/www/mofacts/.env"
scp -i $StageKey .\settingsstaging.json "${StageHost}:/var/www/mofacts/settingsstaging.json"
ssh -i $StageKey $StageHost "chmod 600 /var/www/mofacts/settingsstaging.json"

# Install Memphis SAML cert/key.
scp -i $StageKey "$SamlDir\memphis-saml-public-cert.pem" "${StageHost}:/tmp/memphis-saml-public-cert.pem"
scp -i $StageKey "$SamlDir\memphis-saml-private-key.pem" "${StageHost}:/tmp/memphis-saml-private-key.pem"
ssh -i $StageKey $StageHost "sudo mv /tmp/memphis-saml-public-cert.pem /mofactsAssets_override/memphis-saml-public-cert.pem && sudo mv /tmp/memphis-saml-private-key.pem /mofactsAssets_override/memphis-saml-private-key.pem && sudo chown root:root /mofactsAssets_override/memphis-saml-*.pem && sudo chmod 644 /mofactsAssets_override/memphis-saml-public-cert.pem && sudo chmod 600 /mofactsAssets_override/memphis-saml-private-key.pem"

# Preflight and start the app stack.
ssh -i $StageKey $StageHost "test -f /var/www/mofacts/docker-compose.yaml && test -f /var/www/mofacts/.env && test -f /var/www/mofacts/settingsstaging.json && test -f /mofactsAssets_override/memphis-saml-public-cert.pem && test -f /mofactsAssets_override/memphis-saml-private-key.pem && test -d /dynamic-assets && test -d /h5p-content && test -d /h5p-libraries"
ssh -i $StageKey $StageHost "cd /var/www/mofacts && sudo docker compose --env-file .env -f docker-compose.yaml pull mofacts && sudo docker compose --env-file .env -f docker-compose.yaml up -d --no-build"

# ---------------------------------------------------------------------------
# Production app deploy
# ---------------------------------------------------------------------------

cd C:\dev\MoFaCTS\deploy

$ProdKey = "C:\Users\ppavl\OneDrive\Desktop\prodkey.pem"
$ProdHost = "ubuntu@52.89.109.53"
$SamlDir = "C:\Users\ppavl\OneDrive\Active projects\mofacts-saml"

if (!(Test-Path .\docker-compose.yml)) { throw "Missing production compose file" }
if (!(Test-Path .\.env.production)) { throw "Missing production env file" }
if (!(Test-Path .\settings.json)) { throw "Missing production settings file" }
if (!(Test-Path .\mongo-init\01-create-mofacts-app-user.js)) { throw "Missing Mongo app-user init script" }
if (!(Test-Path "$SamlDir\memphis-saml-public-cert.pem")) { throw "Missing Memphis SAML public cert" }
if (!(Test-Path "$SamlDir\memphis-saml-private-key.pem")) { throw "Missing Memphis SAML private key" }

$RequiredProductionEnv = @(
  "METEOR_SETTINGS_HOST_PATH",
  "ROOT_URL",
  "MONGO_URL",
  "EXPECTED_MONGO_DB_NAME",
  "MONGO_INITDB_ROOT_USERNAME",
  "MONGO_INITDB_ROOT_PASSWORD",
  "MOFACTS_MONGO_APP_USERNAME",
  "MOFACTS_MONGO_APP_PASSWORD",
  "REDIS_URL"
)
$ProductionEnvKeys = Select-String -Path .\.env.production -Pattern "^[A-Za-z_][A-Za-z0-9_]*=" | ForEach-Object { ($_.Line -split "=", 2)[0] }
$MissingProductionEnv = $RequiredProductionEnv | Where-Object { $ProductionEnvKeys -notcontains $_ }
if ($MissingProductionEnv) { throw "Missing production env keys: $($MissingProductionEnv -join ', ')" }
$ProductionMongoUrl = (Select-String -Path .\.env.production -Pattern "^MONGO_URL=" | Select-Object -First 1).Line
if ($ProductionMongoUrl -notmatch "\?[^#]*authSource=MoFACT-meteor3") { throw "Production MONGO_URL must include ?authSource=MoFACT-meteor3" }

# Build and push the production app image.
docker compose --env-file .\.env.production build --no-cache mofacts
docker compose --env-file .\.env.production push mofacts

# Create the runtime filesystem required by the compose mounts.
ssh -i $ProdKey $ProdHost "command -v docker >/dev/null && docker compose version >/dev/null && sudo mkdir -p /var/www/mofacts/mongo-init /mofactsAssets_override /dynamic-assets /h5p-content /h5p-libraries && sudo chown -R ubuntu:ubuntu /var/www/mofacts"

# Copy app runtime inputs.
scp -i $ProdKey .\docker-compose.yml "${ProdHost}:/var/www/mofacts/docker-compose.yaml"
scp -i $ProdKey .\.env.production "${ProdHost}:/var/www/mofacts/.env"
scp -i $ProdKey .\settings.json "${ProdHost}:/var/www/mofacts/settings.json"
scp -i $ProdKey .\mongo-init\01-create-mofacts-app-user.js "${ProdHost}:/var/www/mofacts/mongo-init/01-create-mofacts-app-user.js"
ssh -i $ProdKey $ProdHost "chmod 600 /var/www/mofacts/settings.json"

# Install Memphis SAML cert/key.
scp -i $ProdKey "$SamlDir\memphis-saml-public-cert.pem" "${ProdHost}:/tmp/memphis-saml-public-cert.pem"
scp -i $ProdKey "$SamlDir\memphis-saml-private-key.pem" "${ProdHost}:/tmp/memphis-saml-private-key.pem"
ssh -i $ProdKey $ProdHost "sudo mv /tmp/memphis-saml-public-cert.pem /mofactsAssets_override/memphis-saml-public-cert.pem && sudo mv /tmp/memphis-saml-private-key.pem /mofactsAssets_override/memphis-saml-private-key.pem && sudo chown root:root /mofactsAssets_override/memphis-saml-*.pem && sudo chmod 644 /mofactsAssets_override/memphis-saml-public-cert.pem && sudo chmod 600 /mofactsAssets_override/memphis-saml-private-key.pem"

# Preflight and start the app stack.
ssh -i $ProdKey $ProdHost "test -f /var/www/mofacts/docker-compose.yaml && test -f /var/www/mofacts/.env && test -f /var/www/mofacts/settings.json && test -f /var/www/mofacts/mongo-init/01-create-mofacts-app-user.js && test -f /mofactsAssets_override/memphis-saml-public-cert.pem && test -f /mofactsAssets_override/memphis-saml-private-key.pem && test -d /dynamic-assets && test -d /h5p-content && test -d /h5p-libraries"
ssh -i $ProdKey $ProdHost "cd /var/www/mofacts && sudo docker compose --env-file .env -f docker-compose.yaml pull mofacts && sudo docker compose --env-file .env -f docker-compose.yaml up -d --no-build"
